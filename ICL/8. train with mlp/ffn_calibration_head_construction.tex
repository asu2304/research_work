\begin{convbox}{Convention --- read this first}
PyTorch's \texttt{nn.Linear(in,\,out)} stores weight $W\in\R^{\text{out}\times\text{in}}$ and
computes $y=xW\T$ for a row-vector $x$. \textbf{Every matrix below is shown as $W\T$
(shape $\R^{\text{in}\times\text{out}}$)} --- the matrix $x$ multiplies from the left.
Rows $=$ input dims, columns $=$ output dims. Circuit matrices are
$M_{QK}=(W_Q\T)(W_K\T)\T$ and $M_{OV}=(W_V\T)(W_O\T)\T$.

\smallskip
\textbf{What is new here.} Layers 1--2 are the \emph{same} induction circuit as the earlier
document, but now used as a soft \emph{frequency} head: their output is the in-context
\emph{empirical successor distribution} $\fvec$, not a single argmax token. The contribution of
this document is the position-wise \textbf{FFN head} (one hidden layer, width $h=128$, GELU,
residual) that sits before the unembedding and \emph{re-calibrates} $\fvec$ into the
Bayes-style predictive distribution. It is built only from GELU, scaled identities, and biases.

\smallskip
\textbf{On the GELU $\to$ ReLU trick.} A hinge $\tfrac{1}{g}\,\mathrm{GELU}(g\,u)\to\mathrm{ReLU}(u)$
as the input gain $g\to\infty$; we use $g=200$, already indistinguishable. Every GELU in the head
is such a hardened hinge, so the FFN realises an exact piecewise-linear \emph{step}.
\end{convbox}

\section{Input Embedding}
Each token $v_t\in\{0,\dots,19\}$ at position $t\in\{0,\dots,199\}$ is a concatenated one-hot
\begin{equation}
x_t=\Big[\;\underbrace{\ev{20}{v_t}}_{\text{token }0:20}\;;\;\underbrace{\ev{200}{t}}_{\text{position }20:220}\;\Big]\in\R^{\times220}.
\end{equation}
No learnable embedding. The causal mask added before every softmax is
$M_{t_q,t_k}=-\infty\cdot\one[t_k>t_q]$.

% ════════════════════════════════════════════════════════════════════════════════
\section{Layer 1 --- The Strict Signal Retriever}
\emph{Purpose (inherited).} Suppress all noise keys, forbid self-attention, and write the
identity of the \emph{most recent ruled token strictly before $t$} into dims $220{:}240$.
All four matrices have display shape $\R^{220\times220}$.

\begin{align}
(W^{(1)}_Q)\T &= \sqrt{220}\,I_{220}, &&\text{(scale cancels the }1/\!\sqrt{D}\text{; pattern set by }W_K)\\
(W^{(1)}_K)\T &= T_1\T =
\begin{bmatrix} \zeros_{20\times10} & -10^{9}\one_{20\times10} & \zeros_{20\times200}\\[1pt]
\zeros_{200\times10} & -10^{9}\one_{200\times10} & P\T_{200\times200}\end{bmatrix}\\
(W^{(1)}_V)\T &= \begin{bmatrix} I_{20}&\zeros\\ \zeros&\zeros\end{bmatrix},
&&\text{(pass the token one-hot, drop position)}\\
(W^{(1)}_O)\T &= I_{220}.
\end{align}
The position recency sub-block is $P_{t_k,t_q}=-10^{9}$ if $t_k=t_q$ (no self-attention),
else $-20\,(t_q-t_k)$ (recency).
The score $s_{t_q,t_k}=-10^9\one[v_k\!\ge\!10]-10^9\one[t_k\!=\!t_q]-20(t_q-t_k)\one[v_k\!<\!10]$ makes
the most recent valid ruled key a near-hard argmax. Effective circuits:
$M^{(1)}_{QK}\!\xrightarrow{\div\sqrt{D}}\!T_1\T$, \;
$M^{(1)}_{OV}=\big[\begin{smallmatrix}I_{20}&0\\0&0\end{smallmatrix}\big]$.

\begin{invbox}{Layer-1 key invariant}
For every position $t$ --- ruled or noise --- dims $220{:}240$ hold $\ev{20}{\vstar(t)}$, the
identity of the most recent preceding ruled token, where
$\vstar(t)=v_{t^*},\;t^*=\arg\max\{t_k<t:v_{t_k}<10\}$.
\end{invbox}

% ════════════════════════════════════════════════════════════════════════════════
\section{Layer 2 --- The Frequency-Averaging Induction Head}
\label{sec:l2}
\emph{Purpose.} Look up every position whose predecessor-ruled token equals the current token
$v_t$ --- i.e.\ every past occurrence of the current state $s=v_t$ --- and \emph{average} the
tokens emitted there. The average of those value one-hots is the in-context empirical next-token
distribution given $s$. This is the soft (frequency) use of the induction head; no recency tie-break
is applied, so all matches contribute equally.

\paragraph{Shared matrices.}
\begin{align}
(W^{(2)}_K)\T &=
\begin{bmatrix}\zeros_{10\times20}&\zeros_{10\times200}\\ \gamma\,\mathbf{n}_{10\times20}&\zeros_{10\times200}\\
\zeros_{200\times20}&\zeros_{200\times200}\\ I_{20}&\zeros_{20\times200}\\ \zeros_{200\times20}&\zeros_{200\times200}\end{bmatrix}
\begin{array}{l}\scriptstyle\leftarrow\text{ruled }0{:}10\\\scriptstyle\leftarrow\text{noise }10{:}20\\\scriptstyle\leftarrow\text{position }20{:}220\\\scriptstyle\leftarrow\text{L1 out tok }220{:}240\\\scriptstyle\leftarrow\text{L1 out pos }240{:}440\end{array}
\\[4pt]
(W^{(2)}_V)\T &= \begin{bmatrix}I_{20}&\zeros_{20\times200}\\ \zeros_{420\times20}&\zeros_{420\times200}\end{bmatrix},
\qquad (W^{(2)}_O)\T = I_{220}.
\end{align}
The resulting key is $\;k_{t_k}=\ev{20}{\vstar(t_k)}+\gamma\,\one[v_{t_k}\!\ge\!10]\,\ev{220}{10}$\,;
$\;W_V$ copies the original token one-hot of the attended position.
$W_K$ reads the L1 predecessor identity at $t_k$ and adds $\gamma=-5\times10^4$ on any noise
position (annihilating it). $W_V$ copies the \emph{original} token at the attended position. The query
matches the \emph{current} token against those predecessor identities (analogous to the
\textsc{Green} query, columns $0{:}10$ on dims $0{:}10$ with a noise-trigger row), so the matched
set is $\{t_k:\vstar(t_k)=v_t,\ v_{t_k}\!<\!10\text{ or noise-filtered}\}$.

\paragraph{What Layer 2 writes.} With all matched keys tied in score, softmax is uniform over them
and the output is their value average:
\begin{equation}
[a^{(2)}_t]_{0:20}=\fvec_t,\qquad
[\fvec_t]_i=\frac{\#\{t_k:\vstar(t_k)=v_t,\ v_{t_k}=i\}}{\#\{t_k:\vstar(t_k)=v_t\}}
=\widehat{P}(\text{next}=i\mid s=v_t),
\end{equation}
the \textbf{empirical successor frequency}. Its signal part $[\fvec_t]_{0:10}$ concentrates on the
true successors of $s$ ($\approx\alpha/2$ each for the noisy-Markov task); its noise part
$[\fvec_t]_{10:20}$ holds the empirical noise frequencies (which the head below \emph{discards} in
favour of the exact floor). $\fvec_t$ lands in stream dims $440{:}460$.

\begin{takebox}{Why a calibration head is needed}
Reading $\fvec_t$ out directly with a hot softmax (the earlier $\sigma=50$ unembedding) collapses
onto $\arg\max_i[\fvec_t]_i$ --- an over-confident single-token guess. But the truth is
\emph{two} equiprobable successors at $\tfrac{\alpha}{2}$ each plus a noise floor. The FFN head
restores that calibrated shape, which is exactly what a trained MLP learns.
\end{takebox}

% ════════════════════════════════════════════════════════════════════════════════
\section{The FFN Calibration Head}
\label{sec:ffn}
The position-wise FFN is $\;\tilde h_t = h_t + W_2^{\mathrm{ff}}\,\mathrm{GELU}\!\big(W_1^{\mathrm{ff}}h_t+b_1\big)+b_2$,
with $W_1^{\mathrm{ff}}\in\R^{128\times660}$, $W_2^{\mathrm{ff}}\in\R^{660\times128}$. It does two
things: (i) a per-signal-token \textbf{successor detector} (a hard GELU step on $\fvec$), and
(ii) routing of that detector into a zeroed stream block. The calibrated \emph{levels} are then
applied by the unembedding (\S\ref{sec:unembed}). We use $h=20$ of the $128$ hidden units
(two per ruled token); the rest are zero.

\subsection{Detector geometry}
For a ruled token $i\in\{0,\dots,9\}$ define the soft step on its frequency $c=[\fvec_t]_i$:
\begin{equation}
\mathrm{step}(c)=\frac{\mathrm{relu}(c-\tau)-\mathrm{relu}(c-\tau-w)}{w}=
\begin{cases}0 & c\le\tau\\ (c-\tau)/w & \tau<c<\tau+w\\ 1 & c\ge\tau+w\end{cases}
\;\in[0,1].
\end{equation}
Because an \emph{unobserved} successor has $[\fvec_t]_i=0$ exactly (it never followed $s$) while an
observed one has $[\fvec_t]_i\ge 1/(\text{\#matches})$, any $\tau$ in the gap separates them, and
$\mathrm{step}\to\one[i\text{ was a successor of }s]$.

\subsection{$\;(W_1^{\mathrm{ff}})\T\in\R^{660\times128}$ --- two hinges per ruled token}
Hidden units are paired: unit $2i$ and $2i{+}1$ both read frequency coordinate $440{+}i$ with gain
$g$, offset by $\tau$ and $\tau{+}w$:
\begin{equation}
[(W_1^{\mathrm{ff}})\T]_{\,440+i,\;2i}=g,\quad
[(W_1^{\mathrm{ff}})\T]_{\,440+i,\;2i+1}=g,\qquad
[b_1]_{2i}=-g\tau,\quad [b_1]_{2i+1}=-g(\tau+w),
\end{equation}
for $i=0,\dots,9$; all other entries zero. After GELU, unit $2i$ holds
$\tfrac1g\mathrm{GELU}(g(c-\tau))\!\approx\!\mathrm{relu}(c-\tau)$ and unit $2i{+}1$ holds
$\approx\mathrm{relu}(c-\tau-w)$ (the explicit $1/g$ is folded into $W_2^{\mathrm{ff}}$ below).

\subsection{$\;(W_2^{\mathrm{ff}})\T\in\R^{128\times660}$ --- assemble the step, write to the zero block}
Unit pair $(2i,2i{+}1)$ is differenced and scaled by $1/(g\,w)$, and written into stream dim
$240{+}i$ (inside the L1 zero block $240{:}440$):
\begin{equation}
[(W_2^{\mathrm{ff}})\T]_{\,2i,\;240+i}=\frac{1}{g\,w},\qquad
[(W_2^{\mathrm{ff}})\T]_{\,2i+1,\;240+i}=-\frac{1}{g\,w},\qquad b_2=\zeros .
\end{equation}
The residual add then deposits the detector into the zero block while leaving everything else
untouched:
\begin{equation}
[\tilde h_t]_{240+i}=\underbrace{0}_{\text{stream}}+\;\mathrm{step}\big([\fvec_t]_i\big)
\;=\;\mathbf{s}_i\;\approx\;\one[i\text{ was a successor of }v_t],\qquad i=0,\dots,9.
\end{equation}

\begin{invbox}{FFN invariant}
After the head, dims $240{:}250$ of $\tilde h_t$ hold a $\{0,1\}$ indicator $\mathbf{s}_t$ of
\emph{which ruled tokens were observed as successors of the current state}. The noise frequencies
in $440{:}460$ are deliberately \emph{not} read --- the noise level is supplied analytically below.
\end{invbox}

% ════════════════════════════════════════════════════════════════════════════════
\section{Final Unembedding --- Calibrated Levels}
\label{sec:unembed}
The unembedding reads only the detector block and assigns log-probability \emph{levels}; the noise
logits come entirely from the bias. With $\ell_{\mathrm{on}}=\log(\alpha/2)$,
$\ell_{\mathrm{nz}}=\log\frac{1-\alpha}{V-V_r}$, $\ell_{\mathrm{off}}$ tuned:
\begin{equation}
[(W_{\mathrm{fc}})\T]_{\,240+i,\;i}=\ell_{\mathrm{on}}-\ell_{\mathrm{off}}\;\;(i<V_r),\qquad
[b_{\mathrm{fc}}]_i=
\begin{cases}\ell_{\mathrm{off}} & i<V_r\ (\text{ruled})\\[2pt]\ell_{\mathrm{nz}} & i\ge V_r\ (\text{noise})\end{cases}
\end{equation}
all other weight entries zero (in particular the frequency block $440{:}460$ is \emph{not} read).
The logit vector is therefore
\begin{equation}
z_t[i]=
\begin{cases}
\ell_{\mathrm{off}}+(\ell_{\mathrm{on}}-\ell_{\mathrm{off}})\,\mathbf{s}_i
=\begin{cases}\log(\alpha/2)&\mathbf{s}_i=1\\ \ell_{\mathrm{off}}&\mathbf{s}_i=0\end{cases} & i<V_r\\[10pt]
\log\!\big(\tfrac{1-\alpha}{V-V_r}\big) & i\ge V_r.
\end{cases}
\end{equation}
Because the levels are \emph{log-probabilities}, no temperature is needed: $\mathrm{softmax}(z_t)$
is already the calibrated predictive distribution. When the two successors of $s$ have both been
observed ($\sum_i\mathbf{s}_i=2$, the typical case),
\begin{equation}
P(\hat v=i)=
\begin{cases}\alpha/2 & i\in\{\text{the 2 observed successors}\}\\
(1-\alpha)/(V-V_r) & i\in\{\text{noise}\}\\
\approx 0 & \text{otherwise,}\end{cases}
\end{equation}
i.e.\ exactly the context-Bayes predictor. The finite $\ell_{\mathrm{off}}$ is a small floor that
hedges the rare case where a successor has not yet appeared.

\begin{takebox}{Design takeaway}
The attention is unchanged --- it only \emph{measures} the in-context successor frequency $\fvec$.
All of the improvement over a plain read-out lives in three scalar facts the head bakes in:
\begin{enumerate}[nosep,leftmargin=1.4em]
\item \textbf{Detect, don't weight.} A hard GELU step turns $\fvec$ into a $\{0,1\}$ successor
indicator, so the two true successors are \emph{equalised} to $\tfrac{\alpha}{2}$ each instead of
being split by noisy counts.
\item \textbf{Pin the noise floor.} The noise logits are the closed-form $\log\frac{1-\alpha}{V-V_r}$
(a bias), discarding the noisy empirical noise frequencies entirely.
\item \textbf{Levels are log-probs.} Setting $\ell_{\mathrm{on}}=\log\frac{\alpha}{2}$ removes any
need for a temperature; the softmax output \emph{is} the Bayes-style distribution.
\end{enumerate}
Result: a fully hand-set FFN (identities, GELU hinges, biases --- no trained weights) reproduces
the cross-entropy of the GD-trained minimal MLP.
\end{takebox}
